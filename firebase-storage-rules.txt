# Règles Firebase Storage corrigées
# À copier dans Firebase Console > Storage > Rules

rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    // Règles pour les produits
    match /produits/{productName}/{fileName} {
      allow read: if true;
      allow write: if request.resource.size < 5 * 1024 * 1024 // Max 5MB
        && request.resource.contentType.matches('image/.*');
      allow delete: if true;
    }
    
    // Règles pour les ventes
    match /ventes/{venteNumber}/{fileName} {
      allow read: if true;
      allow write: if request.resource.size < 5 * 1024 * 1024 // Max 5MB
        && request.resource.contentType.matches('image/.*');
      allow delete: if true;
    }
    
    // Règles pour les achats de matériel
    match /Achats/Materiel/{referenceAchat}/{fileName} {
      allow read: if true;
      allow write: if request.resource.size < 5 * 1024 * 1024 // Max 5MB
        && request.resource.contentType.matches('image/.*');
      allow delete: if true;
    }
    
    // Bloquer tous les autres chemins
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
